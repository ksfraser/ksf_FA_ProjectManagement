### 2. Test Cases

#### UT-PROJECT-001-001-001-01: Successful Contract Creation

```php
/**
 * Test successful contract creation with all required fields.
 *
 * @BABOK Related: UT-PROJECT-001-001-001
 * @since 1.0.0
 */
public function testCreateContractSuccess(): void
{
    $data = [
        'contract_ref' => 'CONTRACT-2024-001',
        'customer_id' => 123,
        'description' => 'Custom software development project',
        'total_value' => 50000.00,
        'start_date' => '2024-01-15',
        'end_date' => '2024-06-30',
    ];

    $expectedEntity = new ContractEntity();
    $expectedEntity->contractId = 1;
    $expectedEntity->contractRef = 'CONTRACT-2024-001';
    $expectedEntity->customerId = 123;
    $expectedEntity->description = 'Custom software development project';
    $expectedEntity->totalValue = 50000.00;
    $expectedEntity->startDate = '2024-01-15';
    $expectedEntity->endDate = '2024-06-30';
    $expectedEntity->status = 'draft';

    $this->mockRepository
        ->expects($this->once())
        ->method('create')
        ->with($this->callback(function (ContractEntity $entity) use ($data) {
            return $entity->contractRef === $data['contract_ref']
                && $entity->customerId === $data['customer_id']
                && $entity->description === $data['description']
                && $entity->totalValue === $data['total_value']
                && $entity->status === 'draft';
        }))
        ->willReturn(1);

    $this->mockRepository
        ->expects($this->once())
        ->method('findById')
        ->with(1)
        ->willReturn($expectedEntity);

    $result = $this->service->create($data);

    $this->assertTrue($result['success']);
    $this->assertEquals(1, $result['contract_id']);
    $this->assertEmpty($result['errors']);
}
```

#### UT-PROJECT-001-001-001-12: Default Status is Draft

```php
/**
 * Test that contract status defaults to 'draft'.
 *
 * @BABOK Related: UT-PROJECT-001-001-001
 * @since 1.0.0
 */
public function testCreateContractDefaultStatus(): void
{
    $data = [
        'contract_ref' => 'CONTRACT-2024-001',
        'customer_id' => 123,
        'description' => 'Test contract',
        'total_value' => 50000.00,
        'start_date' => '2024-01-15',
    ];

    $capturedEntity = null;

    $this->mockRepository
        ->expects($this->once())
        ->method('create')
        ->with($this->callback(function (ContractEntity $entity) use (&$capturedEntity) {
            $capturedEntity = $entity;
            return $entity->status === 'draft';
        }))
        ->willReturn(1);

    $this->mockRepository
        ->expects($this->once())
        ->method('findById')
        ->willReturn(new ContractEntity());

    $this->service->create($data);

    $this->assertEquals('draft', $capturedEntity->status);
}
```
